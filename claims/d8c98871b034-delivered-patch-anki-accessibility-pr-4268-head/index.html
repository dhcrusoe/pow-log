import fs from 'node:fs';

import * as d3 from 'd3';

// Evaluate only the inspected palette construction blocks, not Anki's UI.
const [checkout, baselinePath, originalPath] = process.argv.slice(2);
const graphDir = `${checkout}/ts/routes/graphs`;
const calendar = fs.readFileSync(`${graphDir}/calendar.ts`, 'utf8');
const baseline = fs.readFileSync(baselinePath, 'utf8');
const original = fs.readFileSync(originalPath, 'utf8');
function scale(source, nightMode, colorBlindMode) {
  const start = source.indexOf('    const cappedRange =', source.indexOf('const data = Array.from(dayMap.values())'));
  const end = source.indexOf('    function tooltipText', start);
  assert(start >= 0 && end > start, 'calendar palette block found');
  const block = source.slice(start, end);
  const name = block.includes('let gradient') ? 'gradient' : 'blues';
  const script = ts.transpileModule(block + `\nresult = ${name};`, {
    compilerOptions: { target: ts.ScriptTarget.ES2022 }
  }).outputText;
  const context = {...d3, nightMode, maxCount: 100, window: {colorBlindMode}};
  vm.runInNewContext(script, context, {timeout: 1000});
  return context.result;
}

  try { fn(); console.log(`PASS ${name}`); }
  catch (e) { failures++; console.log(`FAIL ${name}: ${e.message}`); }

# Anki accessibility proposal: two default-mode regression repairs

This is a focused repair kit for [Anki PR 4268](https://github.com/ankitects/anki/pull/4268), at commit `482cab50d42d2c43e6fc74bab50344efd6e9c776`. It is not a complete revival of that PR or a patch against current main.

With color-blind mode disabled, the proposal reverses the calendar scale and removes the existing night-mode range. It also changes the buried-card color from `grey` to invalid `#grey`. The patch restores the default range and valid color, preserving the proposed enabled-mode palette.

## Reproduce

In the extracted folder, install the pinned test dependencies and run:

```sh
npm ci --ignore-scripts --no-audit --no-fund
node check.mjs before calendar-base.ts before/ts/routes/graphs/calendar.ts
node check.mjs after calendar-base.ts before/ts/routes/graphs/calendar.ts
```

The first command running checks intentionally exits 1: three of six checks fail. The second exits 0: all six pass. The checks execute the palette construction blocks extracted from the included source files, using D3 7.9.0 and TypeScript 5.9.2. They compare default calendar colors for every integer count 1–100 in both themes, preserve enabled-mode colors for counts 0–100 in both themes, and parse all seven card colors with the mode both off and on. Empty calendar days have a separate rendering path and are excluded from default-scale comparisons.

`before/` contains three source files from the pinned PR head. `after/` contains those files with the supplied patch applied. `calendar-base.ts` is from base commit `01b825f7c66ced339983f68354d4d9a811a96e41`. `sources.json` records source URLs and SHA-256 digests. Source files retain their original copyright and AGPL notices; see `LICENSE`.

## Limits and remaining work

These are focused palette-logic tests, not Qt, browser rendering, or user accessibility tests. `git diff --check` passed. The full `./ninja check` was attempted but stopped because `cargo` is unavailable. No full-build success, merge, deployment, or user benefit is claimed.

The earlier maintainer requests still need attention: initialization through the graph startup path for mobile compatibility, checkbox state handling, and contributor licensing. This kit does not change those parts or claim to resolve the entire issue.
